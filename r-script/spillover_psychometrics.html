temp_df <- indico_rt_profile %>% filter( measures == "gaze_duration", value > 0)

profiles <- c("reading", "memory", "cogcontr")
effects <- c("len", "freq", "surp", "prev_len", "prev_freq", "prev_surp")

interactions <- list()
for (prof in profiles) {
  for (effect in effects) {
    message("Processing: ", prof, " × ", effect)
    prof_mu <- mean(temp_df[[prof]], na.rm = TRUE)
    prof_sd <- sd(temp_df[[prof]], na.rm = TRUE)
    prof_vals <- c(prof_mu - 2 * prof_sd, prof_mu - prof_sd, prof_mu, prof_mu + prof_sd, prof_mu + 2 * prof_sd)
    # create sequence across the observed range of the predictor
    effect_seq <- seq(min(temp_df[[effect]], na.rm = TRUE), max(temp_df[[effect]], na.rm = TRUE), length.out = 100)
    # generate prediction grid
    newdata <- expand.grid(effect_value = effect_seq, profile_value = prof_vals)
    # hold other variables at their means
    for (p in setdiff(profiles, prof)) {newdata[[p]] <- mean(temp_df[[p]], na.rm = TRUE)}
    for (x in setdiff(effects, effect)) {newdata[[x]] <- mean(temp_df[[x]], na.rm = TRUE)}
    #rename
    names(newdata)[names(newdata) == "profile_value"] <- prof
    names(newdata)[names(newdata) == "effect_value"] <- effect
    newdata$subj_id <- NA
    newdata$text_id <- NA
    if (anyDuplicated(names(newdata))) {
      stop("Duplicate column names in newdata: ", 
           paste(names(newdata)[duplicated(names(newdata))],collapse = ", "))
    }

    # prediction without raneff
    preds <- posterior_epred(m, newdata = newdata, re_formula = NA)

    #posterior summaries and labels
    newdata <- newdata %>% mutate(
      estimate = apply(preds, 2, mean), sd_pred = apply(preds, 2, sd),
      lower = estimate - sd_pred, upper = estimate + sd_pred,
      # lower = apply(preds, 2, quantile, probs = 0.025), upper = apply(preds, 2, quantile, probs = 0.975),
      profile_var = prof, effect_var = effect, 
      location = if_else(effect %in% c("len", "freq", "surp"), "Local effect", "Spillover effect"))
    interaction_name <- paste(prof, effect, sep = "_x_")
    interactions[[interaction_name]] <- newdata}
}

make_interaction_plot <- function(interaction_df, profile, effects, plot_type = c("local", "spillover")) {
  plot_type <- match.arg(plot_type)
  dfp <- interaction_df %>% filter( profile_var == profile, effect_var %in% effects) %>%
    mutate(
      effect_var = factor(effect_var, levels = effects),
      x = case_when(
        effect_var == "len" ~ len,
        effect_var == "freq" ~ freq,
        effect_var == "surp" ~ surp,
        effect_var == "prev_len" ~ prev_len,
        effect_var == "prev_freq" ~ prev_freq,
        effect_var == "prev_surp" ~ prev_surp,
        TRUE ~ NA_real_
      ),
      profile_level = {
        vals <- .data[[profile]]
        lvls <- sort(unique(vals))
        if (length(lvls) == 5) {
          factor(vals, levels = lvls, labels = c("-2 SD", "-1 SD", "Mean", "+1 SD", "+2 SD"))
        } else { factor(vals)}})

  ggplot(dfp, aes(x = x, y = estimate, color = profile_level, fill = profile_level, group = profile_level)) +
    geom_ribbon(aes(ymin = lower, ymax = upper), alpha = 0.15, color = NA) +
    geom_line(linewidth = 0.9) +
    facet_wrap(~ effect_var, nrow = 1, scales = "free_x",
               labeller = labeller(effect_var = effect_labels), strip.position = "bottom") +
    scale_x_continuous(breaks = function(limits) seq(limits[1], limits[2], length.out = 4),
                       labels = scales::label_number(accuracy = 1)) +
    scale_y_continuous(breaks = function(limits) seq(limits[1], limits[2], length.out = 5),
                       labels = scales::label_number(accuracy = 0.1)) +
    labs(x = NULL, y = "Log Gaze Duration", color = profile_labels[[profile]], fill = profile_labels[[profile]]) +
    theme_minimal() +
    theme(legend.position = "right", legend.title.align = 0.5,
          strip.placement = "outside", panel.spacing.x = unit(0.5, "cm"), 
          axis.line.x = element_line(colour = "black",linewidth = 0.4),
          axis.line.y = element_line(colour = "black",linewidth = 0.4),
          # the effect labels below each panel
          strip.text.x = element_text(face = "bold"),
          # axis tick labels unbolded
          axis.text.x = element_text(face = "plain"), axis.text.y = element_text(face = "plain"),
          # y-axis title bold
          axis.title.y = element_text(face = "bold"),
          #legend title and entries
          legend.title = element_text(face = "bold", hjust = 0.5), 
          legend.text  = element_text(face = "plain", hjust = 0.5),
          panel.grid.major.x = element_line(linewidth = 0.2, colour = "grey85"),
          panel.grid.major.y = element_line(linewidth = 0.2, colour = "grey90")) +
    guides(fill = "none", color = guide_legend(title.position = "top", label.hjust = 0.5))
}

get_bf <- function(fit, parameter, direction, test) {
  res <- bayesfactor_pointnull(fit, direction = direction, parameters = parameter,
                               effects = "fixed", null = 0, verbose = FALSE) %>% as.data.frame()
  res <- res %>% filter(sub("^b_", "", Parameter) == parameter) %>%
    mutate(BF = exp(as.numeric(log_BF)), test = test) %>%
    dplyr::select(-log_BF)
  res}

bf_results <- imap_dfr(models, function(fit, model_name) {
    map_dfr(parameters, function(parameter) {
          #two-sided BF
          bf_two <- get_bf(fit = fit, parameter = parameter, direction = "two-sided", test = "two-sided")
          #one-sided BF in hypothesized direction
          bf_one <- get_bf(fit = fit, parameter = parameter, direction = directions[[parameter]], test = "one-sided")
          bind_rows(bf_two, bf_one)}) %>%
    mutate(model = model_name, prior_sd = unname(prior_sds[model_name]),
           .before = 1)}) %>% arrange(Parameter, test, prior_sd)
